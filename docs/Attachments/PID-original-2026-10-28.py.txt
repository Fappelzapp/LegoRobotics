    if kurvejanein == 2:
        strecke = 100000
    else:
        speed = speed
    while strecke < amount or kurvejanein == 2:
        reststrecke = amount - strecke
        print(hub.imu.heading())
        if kurvejanein == 0:
            if reststrecke < 1:
                speed_local = speed_local
            elif reststrecke < abs(speed / 10):

                speed_local = int((5 * anfahren) * (speed / abs(speed)))
            else:
                speed_local = speed

        g = hub.imu.heading()

            err = g - stp

        derr = err - erro
        erro = err
        errsum += err
        ric = int(kp * err + ki * errsum + kd * derr)
        ric = int(ric * (speed / abs(speed)))
        motors.drive(ric, speed_local)

def motors_kurve1(target_angle, speed, steering):
    MIN_SPEED = 50
    while True:
        current_angle = hub.imu.heading()
        remaining = target_angle - current_angle
        if abs(remaining) <= 3:
            break
        if abs(remaining) < 10:
            drive_speed = max(MIN_SPEED, abs(remaining) / 10 * abs(speed))
        else:
            drive_speed = abs(speed)
        drive_speed *= 1 if speed > 0 else -1
        motors.drive(-steering, -drive_speed)
        wait(20)
    motors.stop()
    print("Endwinkel:", hub.imu.heading())

def motors_curve_gyro(end_angle_deg, radius_mm, max_speed_mm_s):
    axle_track = 120
    outer_radius = radius_mm + axle_track / 2
    inner_radius = radius_mm - axle_track / 2
    ratio = inner_radius / outer_radius
    speed = 0
    accel_step = 50
    dt = 20
    print(f"Starte Kurve: {end_angle_deg}°, Radius {radius_mm} mm, Max-Speed {max_speed_mm_s} mm/s")
    while True:
        current_angle = abs(hub.imu.heading())
        target_angle = abs(end_angle_deg)
        remaining_angle = target_angle - current_angle
        if remaining_angle > target_angle / 2:
            speed = min(speed + accel_step, max_speed_mm_s)
        else:
            remaining_angle = abs(target_angle - current_angle)
            speed = max((remaining_angle / target_angle) ** 0.4 * max_speed_mm_s, 50)
        if end_angle_deg > 0:
            left_speed = speed
            right_speed = speed * ratio
        else:
            left_speed = speed * ratio
            right_speed = speed
        links.run(-left_speed)
        rechts.run(right_speed)
        if current_angle >= target_angle:
            break
        wait(dt)
    links.stop()
    rechts.stop()
    print(f"Kurve beendet bei Gyro-Winkel: {hub.imu.heading():.1f}°")
